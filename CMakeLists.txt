cmake_minimum_required(VERSION 3.24)
project(DiamondII VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DIAMOND_LOCAL_JUCE "Use a local JUCE checkout" OFF)
if(DIAMOND_LOCAL_JUCE)
  add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850 GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(DiamondII COMPANY_NAME Papielguapo PLUGIN_MANUFACTURER_CODE Papi PLUGIN_CODE DiII FORMATS VST3 Standalone PRODUCT_NAME "DIAMOND II" COPY_PLUGIN_AFTER_BUILD FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE)
juce_add_binary_data(DiamondSkin SOURCES assets/skin-reference.png)
target_sources(DiamondII PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(DiamondII PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_include_directories(DiamondII PUBLIC "$<TARGET_PROPERTY:juce::juce_audio_utils,INTERFACE_INCLUDE_DIRECTORIES>")
target_link_libraries(DiamondII PRIVATE DiamondSkin juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
juce_add_console_app(DspVerificationTest PRODUCT_NAME "DspVerificationTest")
target_sources(DspVerificationTest PRIVATE Tests/DspVerificationTest.cpp)
target_link_libraries(DspVerificationTest PRIVATE DiamondII)
juce_add_console_app(GeneratePluginScreenshot PRODUCT_NAME "GeneratePluginScreenshot")
target_sources(GeneratePluginScreenshot PRIVATE Tests/GeneratePluginScreenshot.cpp)
target_link_libraries(GeneratePluginScreenshot PRIVATE DiamondII)
enable_testing()
add_executable(DspReferenceRender Tests/DspReferenceRender.cpp)
add_test(NAME DSP COMMAND DspVerificationTest "${CMAKE_CURRENT_SOURCE_DIR}/Calibration/reference-fixtures.bin.gz")
add_test(NAME Screenshot COMMAND GeneratePluginScreenshot "${CMAKE_CURRENT_SOURCE_DIR}/assets/screenshot.png")
